# @yantrakit/flowchart-react

A React flowchart component with auto-layout, path detection, and interactive features.

## Installation

```bash
npm install @yantrakit/flowchart-react
# or
pnpm add @yantrakit/flowchart-react
# or
yarn add @yantrakit/flowchart-react
```

## Basic Usage

```tsx
import { FlowChart } from '@yantrakit/flowchart-react';
import '@yantrakit/flowchart-react/styles';

const flow = {
  id: 'my-flow',
  name: 'My Flow',
  nodes: [
    { id: 'start', label: 'Start', type: 'start' },
    { id: 'process', label: 'Process Data', type: 'action' },
    { id: 'end', label: 'End', type: 'end' },
  ],
  edges: [
    { from: 'start', to: 'process', type: 'happy' },
    { from: 'process', to: 'end', type: 'happy' },
  ],
};

function App() {
  return (
    <div style={{ height: '600px' }}>
      <FlowChart flow={flow} />
    </div>
  );
}
```

## Interactive Features

### Handle Node Clicks

```tsx
function App() {
  const handleNodeClick = (node) => {
    console.log('Clicked:', node.id, node.label);
  };

  return (
    <FlowChart
      flow={flow}
      onNodeClick={handleNodeClick}
    />
  );
}
```

### Path Selection

```tsx
import { useState } from 'react';

function App() {
  const [selectedPathId, setSelectedPathId] = useState(null);

  return (
    <FlowChart
      flow={flow}
      selectedPathId={selectedPathId}
      onPathChange={setSelectedPathId}
    />
  );
}
```

### Active Node Highlighting

```tsx
function App() {
  const [activeNodeId, setActiveNodeId] = useState('process');

  return (
    <FlowChart
      flow={flow}
      activeNodeId={activeNodeId}
      onNodeClick={(node) => setActiveNodeId(node.id)}
    />
  );
}
```

## Using the usePaths Hook

```tsx
import { usePaths } from '@yantrakit/flowchart-react';

function PathList({ flow }) {
  const { paths, selectedPathId, setSelectedPathId } = usePaths(flow);

  return (
    <ul>
      {paths.map((path) => (
        <li
          key={path.id}
          onClick={() => setSelectedPathId(path.id)}
          style={{ fontWeight: selectedPathId === path.id ? 'bold' : 'normal' }}
        >
          {path.name} ({path.type})
        </li>
      ))}
    </ul>
  );
}
```

## Full Example

```tsx
import { useState } from 'react';
import { FlowChart } from '@yantrakit/flowchart-react';
import '@yantrakit/flowchart-react/styles';

const registrationFlow = {
  id: 'registration',
  name: 'User Registration',
  nodes: [
    { id: 'start', label: 'Start', type: 'start' },
    { id: 'input', label: 'Enter Details', type: 'action' },
    { id: 'validate', label: 'Valid?', type: 'decision' },
    { id: 'create', label: 'Create User', type: 'action' },
    { id: 'error', label: 'Show Errors', type: 'warning' },
    { id: 'success', label: 'Success', type: 'end' },
  ],
  edges: [
    { from: 'start', to: 'input' },
    { from: 'input', to: 'validate' },
    { from: 'validate', to: 'create', type: 'happy', label: 'Yes' },
    { from: 'validate', to: 'error', type: 'warning', label: 'No' },
    { from: 'error', to: 'input', type: 'warning' },
    { from: 'create', to: 'success', type: 'happy' },
  ],
};

function App() {
  const [selectedPathId, setSelectedPathId] = useState(null);
  const [activeNodeId, setActiveNodeId] = useState();

  return (
    <div style={{ height: '100vh' }}>
      <FlowChart
        flow={registrationFlow}
        selectedPathId={selectedPathId}
        onPathChange={setSelectedPathId}
        activeNodeId={activeNodeId}
        onNodeClick={(node) => setActiveNodeId(node.id)}
      />
    </div>
  );
}
```
